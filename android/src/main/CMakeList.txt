# Android-spezifische Bibliotheken finden (für alle Architekturen)
find_library(LOG_LIB log)

# ===================================================================
# 💡 INTELLEGENTE LOGIK: Wähle die richtigen Einstellungen pro Architektur
# ===================================================================

# Prüfe, welche Architektur Android Studio/Gradle gerade bauen möchte
if(ANDROID_ABI STREQUAL "arm64-v8a")
    message(STATUS "Konfiguriere für ARM64 (arm64-v8a)")
    set(OUTPUT_LIB_NAME "llama-cpp-arm64")
    set(ARCH_COMPILE_FLAGS
        -march=armv8-a
        -mtune=cortex-a76
        -O3 -DNDEBUG -fno-finite-math-only -funroll-loops
        -DLM_GGML_USE_CPU
        -DLM_GGML_CPU_GENERIC
    )

elseif(ANDROID_ABI STREQUAL "x86_64")
    message(STATUS "Konfiguriere für x86_64 (64-Bit Emulator)")
    set(OUTPUT_LIB_NAME "llama-cpp-x86_64")
    set(ARCH_COMPILE_FLAGS
        -march=x86-64 -mtune=generic
        -mavx2 -mavx -msse3 -msse -mfma -mf16c
        -O3 -DNDEBUG
        -DLM_GGML_USE_CPU
        -DLM_GGML_CPU_GENERIC
        -DLM_GGML_USE_AVX2
        -DLM_GGML_USE_AVX
        -DLM_GGML_USE_SSE3
        -DLM_GGML_USE_SSE
        -DLM_GGML_USE_FMA
        -DLM_GGML_USE_F16C
    )

# Füge hier bei Bedarf weitere Architekturen hinzu (z.B. armeabi-v7a)

else()
    message(FATAL_ERROR "Unbekannte oder nicht unterstützte Android-Architektur: ${ANDROID_ABI}")
endif()

#  Baue die Bibliothek einmalig mit den oben gesetzten Variablen

# Wende die architektur-spezifischen Compiler-Flags an

# Verlinke die Android-Bibliotheken
target_link_libraries(llama-cpp-lib PRIVATE
    ${LOG_LIB}
    ${ANDROID_LIB}
)

# Setze den finalen Dateinamen
set_target_properties(llama-cpp-lib PROPERTIES
    OUTPUT_NAME ${OUTPUT_LIB_NAME}
